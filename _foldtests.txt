
    // ---------------- #132-3：折叠 UI 入口 ----------------

    /// 构造一个 md 编辑器（标题 + 代码块），供折叠测试驱动。
    fn md_editor(text: &str) -> Editor {
        crate::editor::tests_support::make_editor(text, Some("notes.md"))
    }

    #[test]
    fn heading_level_parses_atx_forms() {
        assert_eq!(heading_level_of("# 一级"), Some(1));
        assert_eq!(heading_level_of("### 三级"), Some(3));
        assert_eq!(heading_level_of("###### 六级"), Some(6));
        assert_eq!(heading_level_of("#"), Some(1), "裸 # 也是标题");
        assert_eq!(heading_level_of("####### 七级"), None, "7 个 # 不是标题");
        assert_eq!(heading_level_of("#无空格"), None, "# 后须有空格");
        assert_eq!(heading_level_of("正文 # 不是标题"), None, "行中 # 不算");
        assert_eq!(heading_level_of(""), None);
    }

    #[test]
    fn fold_current_collapses_heading_block() {
        let mut e = md_editor("# A\npara1\npara2\n# B\nafter");
        e.buffer.set_cursor(p(1, 0)); // 光标在 A 小节内部
        assert!(e.fold_current(), "光标在小节内应折叠该小节（跳到折叠头）");
        let heads = e.display_map.fold_heads();
        assert_eq!(heads, vec![0], "折叠头 = 标题行 0");
        assert_eq!(
            e.display_map.fold_ranges(),
            vec![(0, 3)],
            "区间到下一个同级标题前（#[B] 行 3 可见）"
        );
    }

    #[test]
    fn fold_current_collapses_fenced_code_block() {
        let mut e = md_editor("intro\n```rust\nlet a = 1;\nlet b = 2;\n```\ntail");
        e.buffer.set_cursor(p(2, 0)); // 代码块内部
        assert!(e.fold_current());
        assert_eq!(e.display_map.fold_ranges(), vec![(1, 5)], "含开闭 ``` 行");
    }

    #[test]
    fn fold_current_refuses_plain_paragraph() {
        let mut e = md_editor("普通正文\n第二行\n\n第三段");
        e.buffer.set_cursor(p(0, 0));
        assert!(!e.fold_current(), "非标题非围栏 → 不可折叠（不假装支持）");
        assert!(e.display_map.fold_heads().is_empty());
    }

    #[test]
    fn unfold_current_restores_rows() {
        let mut e = md_editor("# A\npara1\npara2\n# B");
        e.buffer.set_cursor(p(0, 0));
        assert!(e.fold_current());
        assert_eq!(e.display_map.fold_ranges().len(), 1);
        assert!(e.unfold_current(), "展开当前");
        assert!(e.display_map.fold_ranges().is_empty());
    }

    #[test]
    fn fold_all_collapses_top_level_only() {
        // A 小节含子标题 B：顶层只有 A（B 已含在 A 区间内）。
        let mut e = md_editor("# A\n## B\nb-body\n# C\nc-body");
        let n = e.fold_all();
        assert_eq!(n, 2, "顶层两块：[A...], [C...]");
        assert_eq!(e.display_map.fold_ranges(), vec![(0, 3), (3, 5)]);
    }

    #[test]
    fn unfold_all_clears_every_fold() {
        let mut e = md_editor("# A\nbody\n# B\nbody2");
        e.fold_all();
        assert_eq!(e.display_map.fold_ranges().len(), 2);
        assert!(e.unfold_all(), "全部展开返回 true");
        assert!(e.display_map.fold_ranges().is_empty());
        assert!(!e.unfold_all(), "已无折叠 → false（幂等）");
    }

    #[test]
    fn fold_then_restore_by_head_lines() {
        // 持久化语义：存头行号 → 按当前文本重算区间尾。
        let mut e = md_editor("# A\nbody\n# B\nbody2");
        e.fold_all();
        let heads = e.display_map.fold_heads();
        assert_eq!(heads, vec![0, 2]);
        // 模拟「重启」：新编辑器载同一文本，按头行恢复。
        let mut e2 = md_editor("# A\nbody\n# B\nbody2");
        for h in &heads {
            if let Some((s, en)) = e2.foldable_span_at(*h) {
                e2.display_map
                    .fold(&e2.buffer, Point::new(s, 0)..Point::new(en, 0));
            }
        }
        assert_eq!(e2.display_map.fold_ranges(), e.display_map.fold_ranges());
    }

    #[test]
    fn restore_drops_heads_that_no_longer_fold() {
        // 头行漂移/不再是标题 → 该条丢弃（宁缺毋错）。
        let mut e = md_editor("普通正文\n# 真标题\nbody");
        assert_eq!(e.foldable_span_at(0), None, "行 0 不是标题");
        let span = e.foldable_span_at(1);
        assert_eq!(span, Some((1, 3)));
    }

    #[test]
    fn heading_fold_runs_to_document_end() {
        // 末尾标题折叠到文末。
        let mut e = md_editor("# A\nx\n# Z\nz1\nz2");
        e.buffer.set_cursor(p(2, 0));
        assert!(e.fold_current());
        assert_eq!(e.display_map.fold_ranges(), vec![(2, 5)], "到文末");
    }

    #[test]
    fn docx_editor_has_no_fold_semantics() {
        // 只读文档（docx/xlsx）不做区块折叠：无 md 语义。
        let mut e = md_editor("# A\nbody");
        e.docx = None; // 保持 md 态
        assert_eq!(e.foldable_span_at(0), Some((0, 2)));
    }
